Material Load(string n) { return UnityEditor.AssetDatabase.LoadAssetAtPath<Material>(env + "/Materials/" + n + ".mat"); }
Material Mat(string name, Color c, float metallic, float smooth, Color emission, bool transparent = false)
{
    var p = env + "/Materials/" + name + ".mat";
    var m = UnityEditor.AssetDatabase.LoadAssetAtPath<Material>(p);
    if (m == null) { m = new Material(lit); UnityEditor.AssetDatabase.CreateAsset(m, p); }
    m.shader = lit; m.SetColor("_BaseColor", c); m.SetFloat("_Metallic", metallic); m.SetFloat("_Smoothness", smooth);
    if (emission.maxColorComponent > 0f) { m.SetColor("_EmissionColor", emission); m.EnableKeyword("_EMISSION"); m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.BakedEmissive; }
    else { m.DisableKeyword("_EMISSION"); m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.EmissiveIsBlack; }
    if (transparent) { m.SetFloat("_Surface", 1f); m.SetOverrideTag("RenderType", "Transparent"); m.renderQueue = 3000; m.SetInt("_SrcBlend", (int)UnityEngine.Rendering.BlendMode.SrcAlpha); m.SetInt("_DstBlend", (int)UnityEngine.Rendering.BlendMode.OneMinusSrcAlpha); m.SetInt("_ZWrite", 0); m.EnableKeyword("_SURFACE_TYPE_TRANSPARENT"); }
    UnityEditor.EditorUtility.SetDirty(m); return m;
}

// launch screen placeholder texture
{
    int w = 1024, h = 576; var tex = new Texture2D(w, h, TextureFormat.RGB24, false);
    var px = new Color[w * h];
    for (int y = 0; y < h; y++) for (int x = 0; x < w; x++) {
        float t = (float)y / h;
        var c = Color.Lerp(new Color(0.03f, 0.08f, 0.16f), new Color(0.01f, 0.02f, 0.05f), t);
        if (x % 64 == 0 || y % 64 == 0) c += new Color(0.03f, 0.05f, 0.08f);
        if (y == h * 2 / 5) c = new Color(0.15f, 0.35f, 0.25f);
        // pad and tower silhouette
        int cx = w / 2; if (y < h * 2 / 5 && y > h * 2 / 5 - 12 && Mathf.Abs(x - cx) < 120) c = new Color(0.25f, 0.27f, 0.3f);
        if (y >= h * 2 / 5 && y < h * 2 / 5 + 230 && x > cx + 40 && x < cx + 70) c = new Color(0.22f, 0.24f, 0.27f);
        if (y >= h * 2 / 5 && y < h * 2 / 5 + 200 && Mathf.Abs(x - cx) < 18) c = new Color(0.85f, 0.86f, 0.88f);
        if (y >= h * 2 / 5 + 200 && y < h * 2 / 5 + 240 && Mathf.Abs(x - cx) < 18 - (y - (h * 2 / 5 + 200)) * 18 / 40) c = new Color(0.95f, 0.55f, 0.3f);
        if (y >= h * 2 / 5 && y < h * 2 / 5 + 50 && Mathf.Abs(x - cx) >= 18 && Mathf.Abs(x - cx) < 18 + (50 - (y - h * 2 / 5)) * 0.9f) c = new Color(0.95f, 0.55f, 0.3f);
        px[y * w + x] = c;
    }
    tex.SetPixels(px); tex.Apply();
    var tp = env + "/Textures/LaunchScreen_Placeholder.png";
    System.IO.File.WriteAllBytes(System.IO.Path.Combine(projectRoot, tp), tex.EncodeToPNG());
    UnityEditor.AssetDatabase.ImportAsset(tp, UnityEditor.ImportAssetOptions.ForceSynchronousImport);
    var ti = UnityEditor.AssetImporter.GetAtPath(tp) as UnityEditor.TextureImporter; ti.maxTextureSize = 1024; ti.SaveAndReimport();
    UnityEngine.Object.DestroyImmediate(tex);
}

{

GameObject Prop(string id, Vector3 floorPos, float yaw, Transform parent, string onlyChild = null)
{
    var prefab = UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(env + "/Models/" + id + ".fbx");
    var g = (GameObject)UnityEditor.PrefabUtility.InstantiatePrefab(prefab); g.transform.SetParent(parent, false);
    g.transform.rotation = Quaternion.Euler(0f, yaw, 0f);
    if (onlyChild != null) foreach (Transform c in g.transform) c.gameObject.SetActive(c.name == onlyChild);
    var rs = g.GetComponentsInChildren<Renderer>(); Bounds b = new Bounds(); bool first = true;
    foreach (var r in rs) { if (first) { b = r.bounds; first = false; } else b.Encapsulate(r.bounds); }
    g.transform.position = floorPos + new Vector3(0f, -b.min.y, 0f);
    foreach (var r in rs) { var mc = r.gameObject.GetComponent<MeshCollider>(); if (mc == null) { mc = r.gameObject.AddComponent<MeshCollider>(); } }
    UnityEditor.GameObjectUtility.SetStaticEditorFlags(g, staticFlags); foreach (Transform c in g.GetComponentsInChildren<Transform>(true)) UnityEditor.GameObjectUtility.SetStaticEditorFlags(c.gameObject, staticFlags);
    return g;
}
